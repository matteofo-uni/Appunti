# Operazioni sulle singly-linked list
- Cancellazione
  ```cs
  void list_delete(pointer l, pointer x) {
  	if (x == l) {
  		// se x è il primo elemento
  		l = next(x);
  		free(l);
  	} else {
  		while (next(l) != x) {
  			l = next(l);
  		}
  
  		next(l) = next(x);
  		free(x);
  	}
  }
  ```
  - Tempo:
    - Caso migliore: $\Omega(1)$
    - Caso peggiore: $O(n)$

- Ricerca
	```cs
	void search(pointer l, key k) {
		pointer att = l;
		pointer prec = null;
		
		while (key(att) != k && att != null) {
			prec = att;
			att = nect(att);
		}
	}
	```

# Altre operazioni sulle doubly-linked list
- Minimo
	```cs
	pointer min(pointer l) {
		if (l == null) return null;

		pointer pmin = l;
		pointer patt = next(l);

		while (patt != null) {
			if (key(patt) < key(pmin)) {
				pmin = patt;
			}

			patt = next(patt);
		}

		return pmin;
	}
	```
	- Tempo: $\Theta(n)$
- Successore
	```cs
	pointer succ(pointer l, key k) {
		if (l == null) return null;

		pointer psucc = l;
		while (key(psucc) <= k && psucc != null) {
			psucc = next(psucc);
		}

		pointer patt = next(psucc);
		while (patt != null) {
			if (key(patt) < key(psucc) && key(patt) > k) {
				psucc = patt;
			}

			patt = next(patt);
		}

		return psucc;
	}
	```

# Esercizi
- Siano due listi semplici L1 e L2 contenenti elementi interi. Creare una lista L3 che contenga gli elementi L1 e L2.
	```cs
	pointer union(pointer 7tyuutghjjbn l1, pointer l2) {
		pointer l3 = l1;
		pointer att = l3;

		while (next(att) != null) {
			att = next(att);
		}

		next(att) = l2;

		l1 = null;
		l2 = null;

		return l3;
	}
	```

- Data una lista doppia non circolare di caratteri, verificare se la lista è palindroma.
	```cs
	bool ispalindrome(pointer l) {
		if (pi == null) return true;

		pointer pi = l;
		pointer pf = l;

		while (next(pf) != null) {
			pf = next(pf);
		}

		while (key(pi) == key(pf) && pi != pf && next(pf) != pi) {
			pi = next(pi);
			pf = prev(pf);
		}

		if (pi == pf || next(pf) == pi) return true;
		else return false;
	}
	```
	Tempo: $\Theta(n)$

- Eliminare tutte le occorrenze della chiave K dalla lista L
	```cs
	void delete(pointer l, key k) {
		pointer att = l;
		pointer pre = null;

		while (att != null && key(att) == k) {
			l = next(att);
			free(att);
			att = l;
		}

		while (att != null) {
			if (ket(att) == k) {
				next(pre) = next(att);
				free(att);
				att = next(pre);
			} else {
				pre = att;
				att = next(att);
			}
		}
	}
	```
